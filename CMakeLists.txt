cmake_minimum_required(VERSION 3.20)
project(RoboPack VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(robopack_core
    src/protocol.cpp src/adapters.cpp src/config.cpp src/ranging.cpp src/mapping.cpp src/navigation.cpp src/replay.cpp
    controls/steering.cpp controls/throttle.cpp controls/planner.cpp)
target_include_directories(robopack_core PUBLIC include controls)
add_library(robopack_simulation src/simulation.cpp)
target_link_libraries(robopack_simulation PUBLIC robopack_core)
add_library(robopack_stm32 firmware/stm32/drive_guard.c)
target_include_directories(robopack_stm32 PUBLIC firmware/stm32)
target_link_libraries(robopack_core PUBLIC robopack_stm32)
add_executable(robopack src/main.cpp)
target_link_libraries(robopack PRIVATE robopack_core robopack_simulation)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target robopack_core robopack_simulation robopack_stm32 robopack)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(navigation_tests tests/navigation_tests.cpp)
    target_link_libraries(navigation_tests PRIVATE robopack_core robopack_simulation robopack_stm32)
    foreach(group ranging adapters mapping planner detour navigation protocol replay)
        add_test(NAME ${group} COMMAND navigation_tests ${group})
    endforeach()
    add_executable(firmware_tests tests/firmware_tests.c)
    target_link_libraries(firmware_tests PRIVATE robopack_stm32)
    add_test(NAME stm32_guard COMMAND firmware_tests)
    add_test(NAME demo COMMAND robopack --demo --steps 70 --config ${CMAKE_SOURCE_DIR}/config/robot.conf)
    add_test(NAME replay_roundtrip COMMAND ${CMAKE_COMMAND}
        -DAPP=$<TARGET_FILE:robopack> -DCONFIG=${CMAKE_SOURCE_DIR}/config/robot.conf
        -P ${CMAKE_SOURCE_DIR}/tests/replay_roundtrip.cmake)
endif()

option(ROBOPACK_WITH_C1 "Build stationary C1 capture using an external SLAMTEC SDK" OFF)
set(RPLIDAR_SDK_ROOT "" CACHE PATH "Path to the built official rplidar_sdk checkout")
if(ROBOPACK_WITH_C1)
    find_path(RPLIDAR_INCLUDE sl_lidar_driver.h HINTS "${RPLIDAR_SDK_ROOT}/sdk/include" REQUIRED)
    find_library(RPLIDAR_LIBRARY NAMES sl_lidar_sdk rplidar_sdk
        HINTS "${RPLIDAR_SDK_ROOT}/output/Linux/Release" "${RPLIDAR_SDK_ROOT}/output/Linux/Debug" REQUIRED)
    find_package(Threads REQUIRED)
    add_executable(robopack_c1_capture adapters/c1_capture.cpp)
    target_include_directories(robopack_c1_capture PRIVATE "${RPLIDAR_INCLUDE}")
    target_link_libraries(robopack_c1_capture PRIVATE robopack_core "${RPLIDAR_LIBRARY}" Threads::Threads)
endif()
